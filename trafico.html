function procesarTextoAdaptativo() { const rawText = document.getElementById('input-datos').value; const tbody = document.getElementById('cuerpo-tabla'); if (!rawText.trim()) { alert("Pega primero la información en el cuadro."); return; } // Separar bloques por el patrón de viaje/nave (ej. HLC2026, MAE2026, CMA2026, etc.) const patronSeparador = /(?=[A-Z0-9]{3}2026\d{3})/g; const bloques = rawText.split(patronSeparador); let htmlFilas = ''; let contador = 0; bloques.forEach(bloque => { if (bloque.startsWith("NaveViaje") || bloque.length < 15) return; // 1. Detección exhaustiva de estados de muelle/operación const textoMayus = bloque.toUpperCase(); let statusOficial = "DESCONOCIDO"; if (textoMayus.includes("DEPARTED") || textoMayus.includes("SAILED")) { statusOficial = "DEPARTED"; } else if ( textoMayus.includes("WORKING") || textoMayus.includes("BERTHED") || textoMayus.includes("AT BERTH") || textoMayus.includes("OPERATING") || textoMayus.includes("ALONGSIDE") || textoMayus.includes("MOORED") ) { statusOficial = "WORKING"; } else if (textoMayus.includes("TO ARRIVE") || textoMayus.includes("ANCHORED") || textoMayus.includes("WAITING")) { statusOficial = "TO ARRIVE"; } // 2. Extraer todas las fechas YYYY-MM-DD HH:MM:SS de este bloque const fechas = bloque.match(/\d{4}-\d{2}-\d{2}\s\d{2}:\d{2}:\d{2}(\.\d)?/g) || []; // 3. Extraer Nombre de Nave y Viaje const primeraFechaIdx = bloque.search(/\d{4}-\d{2}-\d{2}/); let encabezadoNave = primeraFechaIdx !== -1 ? bloque.substring(0, primeraFechaIdx).trim() : "Nave"; let claseFila = 'espera'; let badge = 'POR ARRIBAR'; let etb = "-", etd = "-", ata = "-", atd = "-"; // Asignación inteligente por estado detectado if (statusOficial === "WORKING") { claseFila = 'trabajando'; badge = 'OPERANDO'; // En operación: Las primeras dos fechas son CutOff/ETB/ETD y la subsiguiente es el ATA (Atraco real) etb = fechas[2] || fechas[0] || "-"; etd = fechas[3] || fechas[1] || "-"; ata = fechas[4] || "En Muelle"; atd = "-"; } else if (statusOficial === "DEPARTED") { claseFila = 'finalizado'; badge = 'ZARPÓ'; etb = fechas[2] || "-"; etd = fechas[3] || "-"; ata = fechas[4] || "-"; atd = fechas[5] || "-"; } else { // TO ARRIVE claseFila = 'espera'; badge = 'POR ARRIBAR'; etb = fechas[2] || fechas[0] || "-"; etd = fechas[3] || fechas[1] || "-"; ata = "-"; atd = "-"; } contador++; htmlFilas += ` ${badge} ${encabezadoNave} ${etb}
${etd} ${ata} ${atd} ${statusOficial} `; }); if (contador > 0) { tbody.innerHTML = htmlFilas; } else { tbody.innerHTML = 'No se detectaron registros válidos. Pega la selección de la web nuevamente.'; } }
